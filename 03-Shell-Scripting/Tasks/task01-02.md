# Challenge 1 – User Account Management

## 1. Objective

Create a Bash script named user_management.sh that manages Linux user accounts using command-line arguments.

The script should support:

- Create user accounts.
- Delete existing user accounts.
- Reset user passwords.
- List usernames and their UIDs.
- Display help and usage information.

## 2. Available Options

| Option | Description |
|---|---|
| -c, --create | Create a new user |
| -d, --delete | Delete an existing user |
| -r, --reset | Reset user password |
| -l, --list | List usernames and UIDs |
| -h, --help | Display help information |

## 3. Create the Script

Command:

    vim user_management.sh

## 4. Complete Solution

    #!/bin/bash

    # User Management Script
    # Supports create, delete, reset, list and help operations.

    # Function to display help
    show_help() {
        echo "Usage: $0 [OPTION]"
        echo "Options:"
        echo "  -c, --create    Create a new user"
        echo "  -d, --delete    Delete an existing user"
        echo "  -r, --reset     Reset user password"
        echo "  -l, --list      List all users and UIDs"
        echo "  -h, --help      Display help information"
    }

    # Function to create user
    create_user() {
        read -rp "Enter username: " username

        if id "$username" &>/dev/null; then
            echo "Error: Username already exists!"
            return 1
        fi

        sudo useradd -m "$username"

        if [ $? -eq 0 ]; then
            echo "User $username created successfully."
            sudo passwd "$username"
        else
            echo "Error: User creation failed."
            return 1
        fi
    }

    # Function to delete user
    delete_user() {
        read -rp "Enter username to delete: " username

        if ! id "$username" &>/dev/null; then
            echo "Error: User does not exist!"
            return 1
        fi

        if [ "$username" = "root" ]; then
            echo "Error: Root account cannot be deleted."
            return 1
        fi

        read -rp "Are you sure? (yes/no): " confirmation

        if [ "$confirmation" = "yes" ]; then
            sudo userdel "$username"

            if [ $? -eq 0 ]; then
                echo "User $username deleted successfully."
            else
                echo "Error: User deletion failed."
                return 1
            fi
        else
            echo "Deletion cancelled."
        fi
    }

    # Function to reset password
    reset_password() {
        read -rp "Enter username: " username

        if ! id "$username" &>/dev/null; then
            echo "Error: User does not exist!"
            return 1
        fi

        echo "Enter new password for $username:"
        sudo passwd "$username"

        if [ $? -eq 0 ]; then
            echo "Password updated successfully for $username."
        else
            echo "Error: Password reset failed."
            return 1
        fi
    }

    # Function to list users and UIDs
    list_users() {
        echo "Username                 UID"
        echo "----------------------------"

        awk -F: '{printf "%-24s %s\n", $1, $3}' /etc/passwd
    }

    # Main menu using command-line arguments
    case "$1" in
        -c|--create)
            create_user
            ;;
        -d|--delete)
            delete_user
            ;;
        -r|--reset)
            reset_password
            ;;
        -l|--list)
            list_users
            ;;
        -h|--help)
            show_help
            ;;
        *)
            echo "Invalid option."
            show_help
            exit 1
            ;;
    esac

# Challenge 2 – Automated Backup & Recovery Using Shell Scripting

## 1. Objective

Create a Bash script named backup_with_rotation.sh.

The script should:

- Accept a directory path as a command-line argument.
- Create timestamped backup directories.
- Copy all files from the specified directory.
- Retain only the latest 3 backups.
- Automatically remove older backups.
- Display appropriate success and error messages.

## 2. Create the Script

Command:

    vim backup_with_rotation.sh

## 3. Complete Solution

    #!/bin/bash

    # Automated Backup with Rotation
    # Keeps only the latest 3 backups.

    # Check directory argument
    if [ "$#" -ne 1 ]; then
        echo "Usage: $0 <directory_path>"
        exit 1
    fi

    SOURCE_DIR="${1%/}"

    # Check whether directory exists
    if [ ! -d "$SOURCE_DIR" ]; then
        echo "Error: Directory does not exist."
        exit 1
    fi

    # Prevent backup inside the source directory
    SOURCE_REAL=$(realpath "$SOURCE_DIR")
    SCRIPT_DIR=$(realpath "$(dirname "$0")")

    case "$SCRIPT_DIR/" in
        "$SOURCE_REAL/"*)
            echo "Error: Keep the script outside the source directory."
            exit 1
            ;;
    esac

    # Generate unique timestamp
    TIMESTAMP=$(date +%F_%H-%M-%S-%N)

    BACKUP_DIR="$SOURCE_DIR/backup_$TIMESTAMP"

    # Create backup directory
    mkdir -p "$BACKUP_DIR" || exit 1

    # Copy source contents, excluding backup folders
    if ! find "$SOURCE_DIR" -mindepth 1 -maxdepth 1 \
        ! -name 'backup_*' \
        -exec cp -a -t "$BACKUP_DIR" -- {} +; then

        echo "Backup failed!"
        rm -rf -- "$BACKUP_DIR"
        exit 1
    fi

    echo "Backup created: $BACKUP_DIR"

    # Keep only the latest 3 backups
    mapfile -t BACKUPS < <(
        find "$SOURCE_DIR" -mindepth 1 -maxdepth 1 \
            -type d -name 'backup_*' -printf '%f\n' |
        sort -r
    )

    # Remove older backups
    if [ "${#BACKUPS[@]}" -gt 3 ]; then
        for ((i=3; i<${#BACKUPS[@]}; i++)); do
            rm -rf -- "$SOURCE_DIR/${BACKUPS[$i]}"
            echo "Old backup removed: ${BACKUPS[$i]}"
        done
    fi

    echo "Backup rotation completed successfully."

## 10. Important Concepts Used

- $#: Number of command-line arguments.
- $1: First command-line argument.
- ${1%/}: Removes a trailing slash from the directory path.
- -d: Checks whether a directory exists.
- -ne: Checks whether two numbers are not equal.
- date: Generates timestamps.
- %N: Generates nanoseconds for unique backup names.
- mkdir -p: Creates directories.
- find: Searches for files and directories.
- -mindepth: Sets minimum search depth.
- -maxdepth: Sets maximum search depth.
- cp -a: Copies files while preserving attributes.
- mapfile: Reads command output into an array.
- sort -r: Sorts names in reverse order.
- rm -rf: Removes directories and their contents.
- exit 1: Exits with an error status.

## 11. Backup Rotation Logic

The script follows these steps:

1. Accepts a directory path.
2. Checks whether the directory exists.
3. Generates a unique timestamp.
4. Creates a new backup directory.
5. Copies the original directory contents.
6. Finds all existing backup directories.
7. Sorts backup names in reverse chronological order.
8. Keeps the latest 3 backups.
9. Deletes older backups.
10. Displays a completion message.
